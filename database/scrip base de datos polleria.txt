
-- =====================================================
-- BASE DE DATOS: POLLERIA
-- =====================================================

CREATE DATABASE IF NOT EXISTS bd_polleria
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

USE bd_polleria;


-- =====================================================
-- 1. ROLES
-- =====================================================

CREATE TABLE roles (
    id_rol INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(30) NOT NULL UNIQUE,
    descripcion VARCHAR(150)
);


-- =====================================================
-- 2. USUARIOS
-- =====================================================

CREATE TABLE usuarios (
    id_usuario INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(80) NOT NULL,
    apellido VARCHAR(80) NOT NULL,
    correo VARCHAR(120) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    activo TINYINT(1) NOT NULL DEFAULT 1,
    eliminado TINYINT(1) NOT NULL DEFAULT 0,
    id_rol INT NOT NULL,

    CONSTRAINT fk_usuario_rol
        FOREIGN KEY (id_rol)
        REFERENCES roles(id_rol)
);


-- =====================================================
-- 3. MESAS
-- =====================================================

CREATE TABLE mesas (
    id_mesa INT AUTO_INCREMENT PRIMARY KEY,
    numero INT NOT NULL UNIQUE,
    capacidad INT NOT NULL,
    estado ENUM('LIBRE', 'OCUPADA', 'INACTIVA')
        NOT NULL DEFAULT 'LIBRE',
    eliminado TINYINT(1) NOT NULL DEFAULT 0,

    CONSTRAINT chk_mesa_capacidad
        CHECK (capacidad > 0)
);


-- =====================================================
-- 4. CATEGORIAS DE PRODUCTOS
-- =====================================================

CREATE TABLE categorias_productos (
    id_categoria INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(80) NOT NULL UNIQUE,
    descripcion VARCHAR(200),
    eliminado TINYINT(1) NOT NULL DEFAULT 0
);


-- =====================================================
-- 5. PRODUCTOS
-- =====================================================

CREATE TABLE productos (
    id_producto INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    descripcion VARCHAR(200),
    precio DECIMAL(10,2) NOT NULL,
    disponible TINYINT(1) NOT NULL DEFAULT 1,
    eliminado TINYINT(1) NOT NULL DEFAULT 0,
    id_categoria INT NOT NULL,

    CONSTRAINT chk_producto_precio
        CHECK (precio >= 0),

    CONSTRAINT fk_producto_categoria
        FOREIGN KEY (id_categoria)
        REFERENCES categorias_productos(id_categoria)
);


-- =====================================================
-- 6. ESTADOS DE PEDIDO
-- =====================================================

CREATE TABLE estados_pedido (
    id_estado INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(30) NOT NULL UNIQUE,
    descripcion VARCHAR(150)
);


-- =====================================================
-- 7. PEDIDOS
-- =====================================================

CREATE TABLE pedidos (
    id_pedido INT AUTO_INCREMENT PRIMARY KEY,
    fecha_pedido DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

    tipo_atencion ENUM(
        'MESA',
        'PARA_LLEVAR',
        'DELIVERY'
    ) NOT NULL,

    direccion_delivery VARCHAR(255),
    observacion VARCHAR(250),
    total DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    eliminado TINYINT(1) NOT NULL DEFAULT 0,

    id_mesa INT NULL,
    id_usuario INT NOT NULL,
    id_estado INT NOT NULL,

    CONSTRAINT chk_pedido_total
        CHECK (total >= 0),

    CONSTRAINT fk_pedido_mesa
        FOREIGN KEY (id_mesa)
        REFERENCES mesas(id_mesa),

    CONSTRAINT fk_pedido_usuario
        FOREIGN KEY (id_usuario)
        REFERENCES usuarios(id_usuario),

    CONSTRAINT fk_pedido_estado
        FOREIGN KEY (id_estado)
        REFERENCES estados_pedido(id_estado)
);


-- =====================================================
-- 8. DETALLE DE PEDIDO
-- =====================================================

CREATE TABLE detalle_pedido (
    id_detalle INT AUTO_INCREMENT PRIMARY KEY,
    id_pedido INT NOT NULL,
    id_producto INT NOT NULL,
    cantidad INT NOT NULL,
    precio_unitario DECIMAL(10,2) NOT NULL,
    subtotal DECIMAL(10,2) NOT NULL,
    observacion VARCHAR(200),

    CONSTRAINT chk_detalle_cantidad
        CHECK (cantidad > 0),

    CONSTRAINT chk_detalle_precio
        CHECK (precio_unitario >= 0),

    CONSTRAINT chk_detalle_subtotal
        CHECK (subtotal >= 0),

    CONSTRAINT fk_detalle_pedido
        FOREIGN KEY (id_pedido)
        REFERENCES pedidos(id_pedido),

    CONSTRAINT fk_detalle_producto
        FOREIGN KEY (id_producto)
        REFERENCES productos(id_producto)
);
